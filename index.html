import flet as ft import math # ========================= # COLORES ESTILO CARTA ORIENTAL # ========================= SKY = "#B9D7D2" SKY_DARK = "#6F918B" BG = "#D9C7A3" INK = "#11170F" CARD_GREEN = "#1F3B22" CARD_GREEN_2 = "#2F5A32" CARD_GREEN_3 = "#4F7B3B" GOLD = "#D8B05A" GOLD_DARK = "#7A5924" CREAM = "#F8EBC8" PAPER = "#EBD7A8" BLUE = "#2F8AB8" BLUE_DARK = "#16445A" RED = "#C95A36" RED_DARK = "#5A2418" BLACK_CARD = "#2A2923" SHADOW = "#000000" ACCIONES = [ { "nombre": "Agregar maná", "costo": 2, "icono": "木", "color": CARD_GREEN_3, "btn": CARD_GREEN_2, }, { "nombre": "Robar carta", "costo": 3, "icono": "本", "color": BLUE, "btn": BLUE_DARK, }, { "nombre": "Bonificar", "costo": 4, "icono": "刀", "color": RED, "btn": RED_DARK, }, ] def main(page: ft.Page): page.title = "Contador de acciones del comandante" page.theme_mode = ft.ThemeMode.LIGHT page.bgcolor = SKY page.scroll = ft.ScrollMode.AUTO page.padding = 0 page.window.width = 430 page.window.height = 880 # ========================= # ESTADO DE LA APP # ========================= estado = { "inicial": 12, "enderezados": 12, "girados": 0, } # ========================= # CONTROLES DINÁMICOS # ========================= input_tokens = ft.TextField( value="12", width=120, height=62, text_align=ft.TextAlign.CENTER, text_size=32, color=CREAM, bgcolor=INK, border_color=GOLD, focused_border_color=GOLD, border_width=2, border_radius=8, ) inicial_text = ft.Text( "Inicial: 12", size=18, color=GOLD, weight=ft.FontWeight.BOLD, font_family="Georgia", ) enderezados_num = ft.Text( "12", size=72, weight=ft.FontWeight.BOLD, color=CREAM, font_family="Georgia", text_align=ft.TextAlign.CENTER, ) girados_num = ft.Text( "0", size=72, weight=ft.FontWeight.BOLD, color=CREAM, font_family="Georgia", text_align=ft.TextAlign.CENTER, ) usos_texts = [ ft.Text( "0", size=42, weight=ft.FontWeight.BOLD, color=a["color"], font_family="Georgia", text_align=ft.TextAlign.CENTER, ) for a in ACCIONES ] botones_usar = [] # ========================= # FUNCIONES # ========================= def leer_tokens(): try: return max(0, int(input_tokens.value)) except ValueError: return 0 def calcular_usos(): for i, accion in enumerate(ACCIONES): usos_texts[i].value = str(estado["enderezados"] // accion["costo"]) def actualizar_botones(): for i, boton in enumerate(botones_usar): costo = ACCIONES[i]["costo"] if estado["enderezados"] >= costo: boton.opacity = 1 boton.on_click = usar_accion(i) else: boton.opacity = 0.35 boton.on_click = None def refrescar(): input_tokens.value = str(estado["enderezados"]) inicial_text.value = f"Inicial: {estado['inicial']}" enderezados_num.value = str(estado["enderezados"]) girados_num.value = str(estado["girados"]) calcular_usos() actualizar_botones() page.update() def aplicar_tokens(e=None): valor = leer_tokens() estado["inicial"] = valor estado["enderezados"] = valor estado["girados"] = 0 refrescar() def enderezar_tokens(e=None): estado["enderezados"] = estado["inicial"] estado["girados"] = 0 refrescar() def cambiar_input(delta): def accion(e=None): valor = leer_tokens() valor = max(0, valor + delta) input_tokens.value = str(valor) page.update() return accion def usar_accion(indice): def accion(e=None): costo = ACCIONES[indice]["costo"] if estado["enderezados"] >= costo: estado["enderezados"] -= costo estado["girados"] += costo refrescar() return accion # ========================= # COMPONENTES VISUALES # ========================= def borde(ancho=1.5, color=GOLD_DARK): return ft.Border.all(ancho, color) def panel(content, bgcolor=CARD_GREEN, padding=12, radius=12): return ft.Container( bgcolor=bgcolor, border=borde(2, GOLD_DARK), border_radius=radius, padding=padding, shadow=ft.BoxShadow( spread_radius=1, blur_radius=8, color="#55000000", offset=ft.Offset(0, 4), ), content=content, ) def titulo_banda(texto): return ft.Container( bgcolor=CARD_GREEN, border=borde(2, GOLD), border_radius=12, padding=8, alignment=ft.Alignment.CENTER, content=ft.Row( alignment=ft.MainAxisAlignment.CENTER, controls=[ ft.Text("❀", color=GOLD, size=18), ft.Text( texto, size=18, color=GOLD, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ft.Text("❀", color=GOLD, size=18), ], ), ) def boton_rect(texto, bgcolor, on_click, height=56): return ft.Container( height=height, bgcolor=bgcolor, border=borde(2, GOLD), border_radius=8, alignment=ft.Alignment.CENTER, on_click=on_click, content=ft.Text( texto, size=22, color=CREAM, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ) def boton_cuadrado(texto, on_click): return ft.Container( width=56, height=56, bgcolor=CARD_GREEN, border=borde(2, GOLD), border_radius=8, alignment=ft.Alignment.CENTER, on_click=on_click, content=ft.Text( texto, size=30, color=CREAM, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ) def circulo_icono(texto, color): return ft.Container( width=62, height=62, bgcolor=color, border=borde(3, GOLD), border_radius=40, alignment=ft.Alignment.CENTER, content=ft.Text( texto, size=30, color=CREAM, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ) def fila_accion(i, accion): boton = ft.Container( width=86, height=46, bgcolor=accion["btn"], border=borde(2, GOLD), border_radius=8, alignment=ft.Alignment.CENTER, on_click=usar_accion(i), content=ft.Text( "USAR", size=16, color=CREAM, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ) botones_usar.append(boton) return ft.Container( bgcolor="#102018", border=borde(1.5, GOLD_DARK), border_radius=10, padding=10, content=ft.Row( spacing=10, vertical_alignment=ft.CrossAxisAlignment.CENTER, controls=[ circulo_icono(accion["icono"], accion["btn"]), ft.Column( expand=True, spacing=4, controls=[ ft.Text( f"{i + 1}. {accion['nombre']}", size=19, color=CREAM, weight=ft.FontWeight.BOLD, font_family="Georgia", ), ft.Text( f"Costo: {accion['costo']} tokens", size=16, color=accion["color"], weight=ft.FontWeight.BOLD, font_family="Georgia", ), ], ), ft.Column( width=58, spacing=0, horizontal_alignment=ft.CrossAxisAlignment.CENTER, controls=[ usos_texts[i], ft.Text( "veces", size=13, color=CREAM, font_family="Georgia", ), ], ), boton, ], ), ) def carta_token(titulo, numero_control, color_titulo, fondo, girada=False): carta = ft.Container( width=150, height=190, bgcolor=fondo, border=borde(3, GOLD), border_radius=12, padding=8, content=ft.Stack( controls=[ ft.Container( left=5, top=5, right=5, bottom=5, border=borde(1.5, GOLD_DARK), border_radius=8, ), ft.Container( top=12, left=0, right=0, content=ft.Text( titulo, size=14, color=color_titulo, weight=ft.FontWeight.BOLD, text_align=ft.TextAlign.CENTER, font_family="Georgia", ), ), ft.Container( top=52, left=0, right=0, bottom=38, alignment=ft.Alignment.CENTER, content=numero_control, ), ft.Container( bottom=10, left=0, right=0, content=ft.Text( "❀", size=24, color=color_titulo, text_align=ft.TextAlign.CENTER, ), ), ], ), ) if girada: carta.rotate = ft.Rotate(angle=math.pi / 18) return carta # ========================= # HEADER # ========================= header = ft.Container( padding=16, gradient=ft.LinearGradient( begin=ft.Alignment.TOP_CENTER, end=ft.Alignment.BOTTOM_CENTER, colors=[SKY, BG], ), content=ft.Column( horizontal_alignment=ft.CrossAxisAlignment.CENTER, spacing=8, controls=[ ft.Row( alignment=ft.MainAxisAlignment.CENTER, controls=[ ft.Text("☁", color=CREAM, size=26), ft.Text("兎 宮 本", color=GOLD_DARK, size=22, weight=ft.FontWeight.BOLD), ft.Text("☁", color=CREAM, size=26), ], ), panel( bgcolor="#17301F", padding=14, radius=16, content=ft.Text( "Contador de acciones\ndel comandante", size=31, color=CREAM, weight=ft.FontWeight.BOLD, text_align=ft.TextAlign.CENTER, font_family="Georgia", ), ), ], ), ) # ========================= # TOKENS DISPONIBLES # ========================= tokens_panel = panel( bgcolor="#244728", padding=14, content=ft.Column( horizontal_alignment=ft.CrossAxisAlignment.CENTER, spacing=12, controls=[ ft.Row( alignment=ft.MainAxisAlignment.CENTER, vertical_alignment=ft.CrossAxisAlignment.CENTER, spacing=18, controls=[ boton_cuadrado("−", cambiar_input(-1)), input_tokens, boton_cuadrado("+", cambiar_input(1)), ], ), boton_rect("APLICAR", BLACK_CARD, aplicar_tokens), boton_rect("ENDEREZAR", CARD_GREEN_2, enderezar_tokens), inicial_text, ], ), ) # ========================= # ACCIONES # ========================= acciones_panel = panel( bgcolor="#0B1711", padding=10, content=ft.Column( spacing=8, controls=[ fila_accion(0, ACCIONES[0]), fila_accion(1, ACCIONES[1]), fila_accion(2, ACCIONES[2]), ], ), ) # ========================= # ESTADO DE TOKENS # ========================= estado_panel = panel( bgcolor="#0B1711", padding=12, content=ft.Row( alignment=ft.MainAxisAlignment.SPACE_EVENLY, vertical_alignment=ft.CrossAxisAlignment.CENTER, controls=[ ft.Column( horizontal_alignment=ft.CrossAxisAlignment.CENTER, controls=[ carta_token( "TOKENS\nENDEREZADOS", enderezados_num, CARD_GREEN_3, "#244728", girada=False, ), ], ), ft.Column( horizontal_alignment=ft.CrossAxisAlignment.CENTER, controls=[ carta_token( "TOKENS\nGIRADOS", girados_num, RED, BLACK_CARD, girada=True, ), ], ), ], ), ) # ========================= # LAYOUT FINAL # ========================= page.add( ft.Container( gradient=ft.LinearGradient( begin=ft.Alignment.TOP_CENTER, end=ft.Alignment.BOTTOM_CENTER, colors=[SKY, BG, "#9B7C4A"], ), content=ft.Column( spacing=12, controls=[ header, ft.Container( padding=14, content=ft.Column( spacing=12, controls=[ titulo_banda("TOKENS DISPONIBLES"), tokens_panel, titulo_banda("ACCIONES POSIBLES"), acciones_panel, titulo_banda("ESTADO DE LOS TOKENS"), estado_panel, ft.Container( alignment=ft.Alignment.CENTER, content=ft.Text( "🍁 ✦ 花 ✦ 🍁", size=22, color=GOLD_DARK, ), ), ft.Container(height=20), ], ), ), ], ), ) ) refrescar() ft.run(main)
